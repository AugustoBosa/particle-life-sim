# Particle Life 2D v1 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Página 2D de particle life rodando no navegador, com matriz editável, sliders e deploy automático no GitHub Pages.

**Architecture:** `physics.js` é puro (sem DOM): estado em typed arrays e `step(state, params)` com um kernel só para 2D e 3D (2D = uma camada em z), grid espacial com cópia das posições ordenada por célula, e espaço toroidal [0,1)^dims. `render.js` desenha com Three.js `Points` em câmera ortográfica. `main.js` liga física, render, painel lil-gui (`ui.js`) e editor de matriz em canvas (`matrix-editor.js`).

**Tech Stack:** Vite, Three.js, lil-gui, `node --test` (runner nativo do Node), GitHub Actions + Pages.

**Spec:** `docs/brainstorm.md`

## Global Constraints

- Física na CPU; a assinatura `step(state, params)` não muda quando a GPU entrar.
- Espaço da simulação = cubo unitário [0,1)^dims com wrap.
- Força: `β = 0.3`; `r < β → r/β − 1`; `β ≤ r < 1 → a·(1 − |2r − 1 − β|/(1 − β))`; `r ≥ 1 → 0`.
- Cores: 2–8. Partículas: até 20k. `rMax` máximo no slider: 0.3 (o grid precisa de ≥ 3 células por eixo).
- Vite `base: './'` (caminhos relativos funcionam com qualquer nome de repo no Pages).
- Sem dependências além de `vite`, `three`, `lil-gui`.
- Textos da UI em pt-BR.

- [ ] **Step 4: Implementar `src/physics.js`**

```js
export const BETA = 0.3;

export function force(r, a) {
  if (r < BETA) return r / BETA - 1;
  if (r < 1) return a * (1 - Math.abs(2 * r - 1 - BETA) / (1 - BETA));
  return 0;
}

export function randomMatrix(n) {
  return Float32Array.from({ length: n * n }, () => Math.random() * 2 - 1);
}

export function createState(count, colors, dims) {
  return {
    count, dims,
    pos: Float32Array.from({ length: count * dims }, Math.random),
    vel: new Float32Array(count * dims),
    color: Uint8Array.from({ length: count }, () => Math.floor(Math.random() * colors)),
  };
}

// ponytail: CPU single-thread, ~O(n·vizinhos). Teto medido na Task 1; acima dele,
// o caminho é WebGPU compute (fase GPU do spec), não Web Worker.
export function step(state, params) {
  const { count, dims, pos, vel, color } = state;
  const { matrix, colors, rMax, frictionHalfLife, forceFactor, dt } = params;

  // grid: célula ≥ rMax, ≥ 3 células por eixo para o wrap não visitar a mesma célula duas vezes.
  // 2D usa o mesmo kernel com uma camada só em z.
  const cells = Math.max(3, Math.floor(1 / rMax));
  const layers = dims === 3 ? cells : 1;
  const nCells = cells * cells * layers;
  const cellOf = new Int32Array(count);
  const start = new Int32Array(nCells + 1);
  const cellIdx = (v) => Math.min(cells - 1, Math.floor(v * cells));
  for (let i = 0; i < count; i++) {
    const b = i * dims;
    const cz = dims === 3 ? cellIdx(pos[b + 2]) : 0;
    const c = (cz * cells + cellIdx(pos[b + 1])) * cells + cellIdx(pos[b]);
    cellOf[i] = c;
    start[c + 1]++;
  }
  for (let c = 0; c < nCells; c++) start[c + 1] += start[c];

  // cópia ordenada por célula (xyz, z = 0 em 2D): vizinhos ficam contíguos na memória
  const fill = start.slice(0, nCells);
  const sorted = new Int32Array(count);
  const sp = new Float32Array(count * 3);
  const sc = new Uint8Array(count);
  for (let i = 0; i < count; i++) {
    const s = fill[cellOf[i]]++;
    sorted[s] = i;
    for (let d = 0; d < dims; d++) sp[s * 3 + d] = pos[i * dims + d];
    sc[s] = color[i];
  }

  const rMax2 = rMax * rMax, invR = 1 / rMax, k = rMax * forceFactor * dt;
  const friction = Math.pow(0.5, dt / frictionHalfLife);
  const oz0 = dims === 3 ? -1 : 0, oz1 = dims === 3 ? 1 : 0;

  for (let s = 0; s < count; s++) {
    const x = sp[s * 3], y = sp[s * 3 + 1], z = sp[s * 3 + 2];
    const row = sc[s] * colors;
    let c = cellOf[sorted[s]];
    const cx = c % cells; c = (c / cells) | 0;
    const cy = c % cells;
    const cz = (c / cells) | 0;
    let fx = 0, fy = 0, fz = 0;
    for (let oz = oz0; oz <= oz1; oz++) {
      const nz = (cz + oz + layers) % layers;
      for (let oy = -1; oy <= 1; oy++) {
        const ny = (cy + oy + cells) % cells;
        for (let ox = -1; ox <= 1; ox++) {
          const nc = (nz * cells + ny) * cells + (cx + ox + cells) % cells;
          for (let t = start[nc], end = start[nc + 1]; t < end; t++) {
            if (t === s) continue;
            // mínima imagem no toro
            let dx = sp[t * 3] - x; if (dx > 0.5) dx -= 1; else if (dx < -0.5) dx += 1;
            let dy = sp[t * 3 + 1] - y; if (dy > 0.5) dy -= 1; else if (dy < -0.5) dy += 1;
            let dz = sp[t * 3 + 2] - z; if (dz > 0.5) dz -= 1; else if (dz < -0.5) dz += 1;
            const r2 = dx * dx + dy * dy + dz * dz;
            if (r2 === 0 || r2 >= rMax2) continue;
            const r = Math.sqrt(r2);
            const F = force(r * invR, matrix[row + sc[t]]) / r;
            fx += dx * F; fy += dy * F; fz += dz * F;
          }
        }
      }
    }
    const b = sorted[s] * dims;
    vel[b] = vel[b] * friction + fx * k;
    vel[b + 1] = vel[b + 1] * friction + fy * k;
    if (dims === 3) vel[b + 2] = vel[b + 2] * friction + fz * k;
  }

  // posições só depois de todas as forças: todo mundo usou as posições antigas
  for (let i = 0; i < pos.length; i++) {
    pos[i] += vel[i] * dt;
    pos[i] -= Math.floor(pos[i]);
  }
}
```

```js
import * as THREE from 'three';

export const PALETTE = [0xff4d4d, 0x4dff88, 0x4d8cff, 0xffd24d, 0xd24dff, 0x4dffff, 0xff8c4d, 0xffffff];

export function createRenderer(canvas) {
  const renderer = new THREE.WebGLRenderer({ canvas });
  renderer.setPixelRatio(window.devicePixelRatio);
  const scene = new THREE.Scene();
  const camera = new THREE.OrthographicCamera();
  const points = new THREE.Points(
    new THREE.BufferGeometry(),
    new THREE.PointsMaterial({ size: 3, vertexColors: true, sizeAttenuation: false }),
  );
  points.frustumCulled = false; // as posições mudam todo frame; bounding sphere ficaria velha
  scene.add(points);

  // mostra o quadrado [0,1]² inteiro, centralizado, sem distorcer
  function resize() {
    const w = window.innerWidth, h = window.innerHeight, a = w / h;
    renderer.setSize(w, h);
    const sx = Math.max(a, 1) / 2, sy = Math.max(1 / a, 1) / 2;
    Object.assign(camera, { left: 0.5 - sx, right: 0.5 + sx, top: 0.5 + sy, bottom: 0.5 - sy, near: -1, far: 1 });
    camera.updateProjectionMatrix();
  }
  window.addEventListener('resize', resize);
  resize();

  const col = new THREE.Color();
  return {
    draw(state) {
      let geo = points.geometry;
      if (geo.attributes.position?.count !== state.count) {
        geo.dispose();
        geo = points.geometry = new THREE.BufferGeometry();
        geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(state.count * 3), 3));
        geo.setAttribute('color', new THREE.BufferAttribute(new Float32Array(state.count * 3), 3));
      }
      const p = geo.attributes.position.array, c = geo.attributes.color.array, { dims } = state;
      for (let i = 0; i < state.count; i++) {
        for (let d = 0; d < dims; d++) p[i * 3 + d] = state.pos[i * dims + d];
        col.setHex(PALETTE[state.color[i]]).toArray(c, i * 3);
      }
      geo.attributes.position.needsUpdate = true;
      geo.attributes.color.needsUpdate = true;
      renderer.render(scene, camera);
    },
  };
}
```

- [ ] **Step 1: Criar `src/matrix-editor.js`**

```js
import { PALETTE } from './render.js';

const hex = (n) => '#' + n.toString(16).padStart(6, '0');

// linha i = cor que reage, coluna j = cor que influencia; verde atrai, vermelho repele
export function createMatrixEditor(canvas, params) {
  const ctx = canvas.getContext('2d');
  const S = canvas.width;

  function draw() {
    const n = params.colors, cell = S / (n + 1);
    ctx.clearRect(0, 0, S, S);
    for (let i = 0; i < n; i++) {
      ctx.fillStyle = hex(PALETTE[i]);
      ctx.fillRect((i + 1) * cell, 0, cell, cell);
      ctx.fillRect(0, (i + 1) * cell, cell, cell);
      for (let j = 0; j < n; j++) {
        const v = params.matrix[i * n + j];
        ctx.fillStyle = v > 0 ? `rgba(60,220,90,${v})` : `rgba(230,60,60,${-v})`;
        ctx.fillRect((j + 1) * cell + 1, (i + 1) * cell + 1, cell - 2, cell - 2);
      }
    }
  }

  function edit(e, delta) {
    e.preventDefault();
    const n = params.colors, r = canvas.getBoundingClientRect();
    const j = Math.floor(((e.clientX - r.left) / r.width) * (n + 1)) - 1;
    const i = Math.floor(((e.clientY - r.top) / r.height) * (n + 1)) - 1;
    if (i < 0 || j < 0 || i >= n || j >= n) return;
    const k = i * n + j;
    params.matrix[k] = Math.max(-1, Math.min(1, params.matrix[k] + delta));
    draw();
  }

  canvas.addEventListener('click', (e) => edit(e, 0.2));
  canvas.addEventListener('contextmenu', (e) => edit(e, -0.2));
  draw();
  return { draw };
}
```
